import { useEffect, useMemo, useState } from "react";
import { useNavigate } from "react-router-dom";
import {
  Box,
  Button,
  Card,
  CardContent,
  CardMedia,
  Chip,
  CircularProgress,
  Container,
  Grid,
  Stack,
  Typography,
} from "@mui/material";

import hero from "../../assets/images/hero/hero.jpg";
import gallery1 from "../../assets/images/gallery/gallery1.jpg";
import gallery2 from "../../assets/images/gallery/gallery2.jpg";
import gallery3 from "../../assets/images/gallery/gallery3.jpg";
import gallery4 from "../../assets/images/gallery/gallery4.jpg";

import { toast } from "react-toastify";

import { getRooms } from "../../api/roomApi";
import { getMyBookings } from "../../api/bookingApi";

function Dashboard() {
  const navigate = useNavigate();
  const [loading, setLoading] = useState(true);
  const [rooms, setRooms] = useState([]);
  const [bookings, setBookings] = useState([]);

  const loadDashboard = async () => {
    try {
      setLoading(true);
      const roomResponse = await getRooms();
      const bookingResponse = await getMyBookings();
      setRooms(roomResponse || []);
      setBookings(bookingResponse || []);
    } catch (error) {
      toast.error(
        error.response?.data?.detail ||
          "Unable to load dashboard."
      );
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    loadDashboard();
  }, []);

  const stats = useMemo(
    () => ({
      availableRooms: rooms.filter((room) => room.is_available).length,
      confirmedBookings: bookings.filter(
        (booking) => booking.status === "CONFIRMED"
      ).length,
      pendingBookings: bookings.filter(
        (booking) => booking.status === "PENDING_PAYMENT"
      ).length,
      cancelledBookings: bookings.filter(
        (booking) => booking.status === "CANCELLED"
      ).length,
    }),
    [rooms, bookings]
  );

  const featuredRooms = useMemo(
    () => rooms.filter((room) => room.is_available).slice(0, 3),
    [rooms]
  );

  const recentBookings = useMemo(() => bookings.slice(0, 4), [bookings]);

  if (loading) {
    return (
      <Box
        sx={{
          display: "flex",
          justifyContent: "center",
          alignItems: "center",
          height: "100vh",
          bgcolor: "#F8FAFC",
        }}
      >
        <CircularProgress size={64} />
      </Box>
    );
  }

  return (
    <Box sx={{ bgcolor: "#F5F7FA", minHeight: "100vh" }}>
      <Box sx={{ position: "relative", overflow: "hidden" }}>
        <Box
          component="img"
          src={hero}
          alt="Hotel hero"
          sx={{
            width: "100%",
            height: { xs: 420, md: 520 },
            objectFit: "cover",
            filter: "brightness(0.6)",
          }}
        />
        <Box
          sx={{
            position: "absolute",
            inset: 0,
            background:
              "linear-gradient(180deg, rgba(15,23,42,0.25), rgba(15,23,42,0.9))",
          }}
        />
        <Container
          maxWidth="xl"
          sx={{
            position: "absolute",
            inset: 0,
            display: "flex",
            alignItems: "center",
            justifyContent: { xs: "center", md: "flex-start" },
          }}
        >
          <Box sx={{ color: "#fff", maxWidth: 720, py: 8 }}>
            <Typography
              variant="h2"
              fontWeight={800}
              sx={{ lineHeight: 1.05, mb: 2 }}
            >
              Discover premium stays with
              <Box component="span" sx={{ color: "#FFD54F" }}>
                modern comfort
              </Box>
            </Typography>
            <Typography
              variant="h6"
              color="rgba(255,255,255,0.85)"
              sx={{ mb: 4 }}
            >
              Manage bookings, view room selections, and complete payments with
              a refined hotel experience designed for today’s traveler.
            </Typography>
            <Stack
              direction={{ xs: "column", sm: "row" }}
              spacing={2}
              justifyContent={{ xs: "center", md: "flex-start" }}
            >
              <Button
                variant="contained"
                size="large"
                onClick={() => navigate("/customer/search-rooms")}
                sx={{
                  bgcolor: "#FFD54F",
                  color: "#000",
                  px: 5,
                  py: 1.8,
                  borderRadius: 3,
                  fontWeight: 700,
                  boxShadow: "0 18px 40px rgba(255,213,79,0.18)",
                  "&:hover": {
                    bgcolor: "#FFC107",
                  },
                }}
              >
                Search Rooms
              </Button>
              <Button
                variant="outlined"
                size="large"
                onClick={() => navigate("/customer/booking")}
                sx={{
                  borderColor: "rgba(255,255,255,0.9)",
                  color: "#fff",
                  px: 5,
                  py: 1.8,
                  borderRadius: 3,
                  fontWeight: 700,
                  "&:hover": {
                    borderColor: "#FFD54F",
                    color: "#FFD54F",
                  },
                }}
              >
                My Bookings
              </Button>
            </Stack>
          </Box>
        </Container>
      </Box>

      <Container maxWidth="xl" sx={{ py: 10 }}>
        <Grid container spacing={4}>
          {[
            {
              title: "Available Rooms",
              value: stats.availableRooms,
              accent: "#0F172A",
              subtitle: "Instantly book the best rooms.",
            },
            {
              title: "Confirmed Stays",
              value: stats.confirmedBookings,
              accent: "#2E7D32",
              subtitle: "Keep your trips on track.",
            },
            {
              title: "Pending Payments",
              value: stats.pendingBookings,
              accent: "#ED6C02",
              subtitle: "Complete payments quickly.",
            },
            {
              title: "Cancelled",
              value: stats.cancelledBookings,
              accent: "#D32F2F",
              subtitle: "Review cancellations safely.",
            },
          ].map((item) => (
            <Grid item xs={12} md={3} key={item.title}>
              <Card
                sx={{
                  borderRadius: 4,
                  minHeight: 190,
                  p: 3,
                  boxShadow: "0 22px 48px rgba(15,23,42,0.08)",
                }}
              >
                <Typography
                  variant="subtitle2"
                  color="text.secondary"
                  sx={{ textTransform: "uppercase", letterSpacing: 1.2 }}
                >
                  {item.title}
                </Typography>
                <Typography
                  variant="h3"
                  fontWeight={800}
                  sx={{ mt: 2, color: item.accent }}
                >
                  {item.value}
                </Typography>
                <Typography color="text.secondary" sx={{ mt: 1 }}>
                  {item.subtitle}
                </Typography>
              </Card>
            </Grid>
          ))}
        </Grid>

        <Box sx={{ mt: 10 }}>
          <Stack
            direction={{ xs: "column", md: "row" }}
            justifyContent="space-between"
            alignItems="center"
            spacing={2}
            mb={4}
          >
            <Box>
              <Typography variant="h4" fontWeight={700}>
                Featured rooms for your next stay
              </Typography>
              <Typography color="text.secondary" sx={{ mt: 1 }}>
                Handpicked rooms with premium amenities and beautiful views.
              </Typography>
            </Box>
            <Button
              variant="contained"
              onClick={() => navigate("/customer/search-rooms")}
              sx={{
                borderRadius: 3,
                px: 4,
                py: 1.5,
                bgcolor: "#0F172A",
                color: "#fff",
                fontWeight: 700,
                "&:hover": {
                  bgcolor: "#131F3C",
                },
              }}
            >
              Explore Rooms
            </Button>
          </Stack>

          <Grid container spacing={4}>
            {featuredRooms.length > 0 ? (
              featuredRooms.map((room) => (
                <Grid item xs={12} md={4} key={room.id}>
                  <Card
                    sx={{
                      borderRadius: 4,
                      overflow: "hidden",
                      boxShadow: "0 30px 60px rgba(15,23,42,0.08)",
                      transition: "transform .3s",
                      "&:hover": {
                        transform: "translateY(-6px)",
                      },
                    }}
                  >
                    <CardMedia
                      component="img"
                      height="260"
                      image={
                        room.image_url ||
                        "https://images.unsplash.com/photo-1505693416388-ac5ce068fe85?auto=format&fit=crop&w=900&q=80"
                      }
                      alt={room.room_type}
                    />
                    <CardContent>
                      <Stack
                        direction="row"
                        justifyContent="space-between"
                        alignItems="center"
                        mb={1}
                      >
                        <Typography variant="h6" fontWeight={700}>
                          {room.room_type}
                        </Typography>
                        <Chip
                          label={room.is_available ? "Available" : "Booked"}
                          color={room.is_available ? "success" : "error"}
                          size="small"
                        />
                      </Stack>
                      <Typography color="text.secondary">
                        Room {room.room_number} · Floor {room.floor_number || "—"}
                      </Typography>
                      <Typography color="text.secondary" sx={{ mt: 1 }}>
                        Capacity: {room.capacity} guest{room.capacity === 1 ? "" : "s"}
                      </Typography>
                      <Stack direction="row" spacing={1} flexWrap="wrap" sx={{ mt: 2 }}>
                        {room.ac && <Chip label="AC" size="small" />}
                        {room.wifi && <Chip label="WiFi" size="small" />}
                        {room.tv && <Chip label="TV" size="small" />}
                        {room.breakfast_included && (
                          <Chip label="Breakfast" size="small" />
                        )}
                      </Stack>
                      <Typography
                        variant="h5"
                        fontWeight={700}
                        sx={{ mt: 3, color: "#D4AF37" }}
                      >
                        ₹{room.price}
                      </Typography>
                      <Button
                        fullWidth
                        variant="contained"
                        sx={{
                          mt: 3,
                          borderRadius: 3,
                          textTransform: "none",
                          bgcolor: "#0F172A",
                          color: "#fff",
                          "&:hover": {
                            bgcolor: "#131F3C",
                          },
                        }}
                        onClick={() => navigate(`/customer/room/${room.id}`)}
                      >
                        View Details
                      </Button>
                    </CardContent>
                  </Card>
                </Grid>
              ))
            ) : (
              <Grid item xs={12}>
                <Card sx={{ p: 6, borderRadius: 4, boxShadow: 4 }}>
                  <Typography variant="h6" fontWeight={700}>
                    No featured rooms available.
                  </Typography>
                  <Typography color="text.secondary" sx={{ mt: 1 }}>
                    Try a different search or refresh to see the latest offers.
                  </Typography>
                </Card>
              </Grid>
            )}
          </Grid>
        </Box>

        <Box sx={{ mt: 10, py: 8, px: 4, background: "linear-gradient(135deg, rgba(255,245,157,0.18), rgba(255,255,255,0.9))", borderRadius: 4 }}>
          <Grid container spacing={4} alignItems="center">
            <Grid item xs={12} md={7}>
              <Typography variant="h4" fontWeight={700} mb={2}>
                Your personal booking hub
              </Typography>
              <Typography color="text.secondary" sx={{ mb: 3 }}>
                Keep track of every stay, payment, and itinerary from one elegant dashboard.
              </Typography>
              <Stack direction={{ xs: "column", sm: "row" }} spacing={2}>
                <Button
                  variant="contained"
                  onClick={() => navigate("/customer/booking-history")}
                  sx={{ borderRadius: 3, px: 4, py: 1.5, bgcolor: "#0F172A", color: "#fff", fontWeight: 700 }}
                >
                  View Bookings
                </Button>
                <Button
                  variant="outlined"
                  onClick={() => navigate("/customer/payments")}
                  sx={{ borderRadius: 3, px: 4, py: 1.5, color: "#0F172A", borderColor: "#0F172A" }}
                >
                  Manage Payments
                </Button>
              </Stack>
            </Grid>
            <Grid item xs={12} md={5}>
              <Box
                component="img"
                src={gallery1}
                alt="Hotel preview"
                sx={{ width: "100%", borderRadius: 4, boxShadow: "0 32px 70px rgba(15,23,42,0.08)" }}
              />
            </Grid>
          </Grid>
        </Box>

        <Box sx={{ mt: 10 }}>
          <Grid container spacing={4}>
            {[gallery3, gallery4, gallery2].map((image, index) => (
              <Grid item xs={12} md={4} key={index}>
                <Box
                  component="img"
                  src={image}
                  alt={`Gallery ${index + 1}`}
                  sx={{
                    width: "100%",
                    borderRadius: 4,
                    height: 320,
                    objectFit: "cover",
                    boxShadow: "0 22px 48px rgba(15,23,42,0.1)",
                  }}
                />
              </Grid>
            ))}
          </Grid>
        </Box>
      </Container>
    </Box>
  );
}

export default Dashboard;
